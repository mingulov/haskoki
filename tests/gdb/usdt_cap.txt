commands
silent
printf "NEW ctx=%llx parent=%lld n=%d\n", $_probe_arg0, $_probe_arg1, $_probe_arg3
python
import gdb
n = int(gdb.parse_and_eval("$_probe_arg3"))
base = int(gdb.parse_and_eval("$_probe_arg2"))
inf = gdb.selected_inferior()
for i in range(min(n, 8)):
    e = base + i * 24
    key_ptr = int.from_bytes(inf.read_memory(e, 8), 'little')
    val_ptr = int.from_bytes(inf.read_memory(e + 8, 8), 'little')
    size = int.from_bytes(inf.read_memory(e + 16, 8), 'little', signed=True)
    key = inf.read_memory(key_ptr, 64).tobytes().split(b'\0')[0].decode()
    if size == -1:
        val = inf.read_memory(val_ptr, 128).tobytes().split(b'\0')[0].decode()
        print(f"  {key} = str:{val}")
    elif size == -2:
        print(f"  {key} = word:{val_ptr}")
    else:
        print(f"  {key} = blob(len {size})")
end
continue
end
break -probe-stap crypto_auditing:release_context
